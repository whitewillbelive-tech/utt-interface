async function session(){if(!token){let r;try{r=await fetch('./api/session')}catch{throw networkError()}if(!r.ok)throw new GateError('SESSION_REJECTED','로컬 세션을 열지 못했습니다. 페이지를 새로고침하세요.','authorization');try{token=(await r.json()).session}catch{throw new GateError('SERVER_RESPONSE_INVALID','서버 응답을 읽을 수 없습니다.','response')}}}
async function api(path,opts={}){await session();let r;try{r=await fetch('./api/'+path,{...opts,headers:{'X-UTT-Session':token,...opts.headers}})}catch{throw networkError()}let d;try{d=await r.json()}catch{throw new GateError('SERVER_RESPONSE_INVALID','서버 응답을 읽을 수 없습니다.','response')}if(!r.ok)throw new GateError(d.error_code||d.error||'SERVER_REJECTED',d.message||'서버가 요청을 거부했습니다.',d.stage||'server');return d;}
function preflight(file,path){
 if(!file)throw new GateError('FILE_NOT_SELECTED','파일을 먼저 선택해 주세요.','preflight');
 const type=(file.type||'').toLowerCase();
 if(!/\.zip$/i.test(file.name||'')||!['','application/zip','application/x-zip-compressed','application/x-zip','application/octet-stream'].includes(type))throw new GateError('SELECTED_FILE_TYPE_MISMATCH','ZIP 파일을 선택해 주세요.','preflight');
 if(file.size!==expectedSizes[path])throw new GateError('SELECTED_FILE_SIZE_MISMATCH','선택 파일 크기 불일치: 지정된 원본 ZIP을 선택하세요.','preflight');
}
async function verifyPreviousPreview(){
 if(!loaded||!validation)return {result:'NONE',verified:null};
 try{
  const status=await api('status');
  if(!status.loaded||status.validation?.dataset_id!==validation.dataset_id||status.validation?.zip_sha256!==validation.zip_sha256){loaded=false;return {result:'NOT_RETAINED',verified:false};}
  const found=await api('search?'+new URLSearchParams({mode:'body',q:'솔로몬',limit:'1',group:'0'}));
  const id=found.results?.[0]?.id;
  if(!id||!found.matched_record_count)return {result:'UNVERIFIED',verified:null};
  const row=await api('detail?id='+encodeURIComponent(id));
  if(canonicalView(row.canonical_record).id!==id)return {result:'UNVERIFIED',verified:null};
  return {result:'VERIFIED',verified:true};
 }catch{return {result:'UNVERIFIED',verified:null};}
}
async function reportFailure(e){
 const err=e instanceof GateError?e:new GateError('LOCAL_ERROR','처리 중 오류가 발생했습니다.','client');
 const state=await verifyPreviousPreview();
 lastFileAttempt={error_code:err.error_code,stage:err.stage,previous_preview_retained_verified:state.verified};
 const suffix=state.result==='VERIFIED'?'이전 B 미리보기의 실제 검색과 상세 열람을 확인했습니다.':state.result==='NONE'?'확인할 이전 B 미리보기가 없습니다.':state.result==='NOT_RETAINED'?'이전 B 미리보기가 현재 세션에 없습니다.':'이전 B 미리보기 보존 여부를 확인하지 못했습니다.';
 setStatus(`${err.message} [${err.error_code}] · ${suffix}`);buttons();
}
