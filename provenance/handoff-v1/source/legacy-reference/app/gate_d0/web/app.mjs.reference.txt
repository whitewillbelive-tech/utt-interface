async function request(path){
 try{if(!token){const r=await fetch('./api/session');if(!r.ok)throw Error('세션 오류');token=(await r.json()).session}const r=await fetch('./api/'+path,{headers:{'X-UTT-D0-Session':token}});if(!r.ok)throw Error('로컬 자료 조회 오류');return r;}catch{throw Error('로컬 검토 서버에 연결할 수 없습니다. 실행 창을 확인하세요.')}
}
